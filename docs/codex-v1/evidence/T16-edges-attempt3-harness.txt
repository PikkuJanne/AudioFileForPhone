function Assert-NoReparse {
    param([string]$Path)
    $item = Get-Item -LiteralPath $Path -Force -ErrorAction Stop
    while ($null -ne $item) {
        Assert-True (($item.Attributes -band [IO.FileAttributes]::ReparsePoint) -eq 0) 'Refusing reparse fixture/ancestor.'
        $item = $item.Parent
    }
}
function Get-SafeTree {
    param([string]$Root)
    Assert-NoReparse $Root
    $pending = New-Object 'System.Collections.Generic.Stack[string]'; $items = New-Object 'System.Collections.Generic.List[object]'
    $pending.Push($Root)
    while ($pending.Count -gt 0) {
        foreach ($item in @(Get-ChildItem -LiteralPath $pending.Pop() -Force -ErrorAction Stop)) {
            Assert-True (($item.Attributes -band [IO.FileAttributes]::ReparsePoint) -eq 0) 'Refusing reparse inside exact owned tree.'
            $items.Add($item); if ($item.PSIsContainer) { $pending.Push($item.FullName) }
        }
    }
    return $items.ToArray()
}

function Invoke-OwnedNative {
    param([string]$Tool, [string[]]$NativeArguments, [object]$Parser = $null, [scriptblock]$Callback = $null,
          [object]$Duration = $null, [int]$StdoutLimit = 1048576, [int]$StderrLimit = 65536,
          [object]$Cancellation = $null, [string]$ExpectedStatus = 'Completed', [object]$ExpectedCode = 0)
    $prior = $script:descendantReleaseProved; $script:descendantReleaseProved = $false
    $ownsContext = $null -eq $Cancellation
    $context = if ($ownsContext) { New-AudioCancellationContext } else { $Cancellation }
    $deadline = New-Object T16EdgeSupport.Deadline($context, 180000)
    try {
        $parameters = @{ ExecutablePath = $Tool; Arguments = $NativeArguments; Cancellation = $context; MaxStdoutChars = $StdoutLimit; MaxStderrChars = $StderrLimit }
        if ($null -ne $Parser) { $parameters.ProgressParser = $Parser }
        if ($null -ne $Callback) { $parameters.ProgressCallback = $Callback; $parameters.ProgressDurationSeconds = $Duration }
        $native = Invoke-AudioNativeProcess @parameters
        if ($native.ProcessReleased) { $script:descendantReleaseProved = $prior }
        $report.native_runs += [pscustomobject]@{ executable = $Tool; arguments = @($NativeArguments | ForEach-Object { $_.Replace($testRoot, '<OWNED_TEMP>') }); status = $native.Status; exit_code = $native.ExitCode; process_released = $native.ProcessReleased; deadline_requested = $deadline.TimedOut; duration_ms = $native.DurationMilliseconds }
        Assert-True (-not $deadline.TimedOut -and $native.Status -ceq $ExpectedStatus -and $native.ExitCode -eq $ExpectedCode -and $native.ProcessReleased) ('Native fixture/check failed: ' + $native.Status + '/' + $native.ExitCode + ' ' + $native.Error + ' ' + $native.Stderr)
        return $native
    } finally { $deadline.Dispose(); if ($ownsContext) { $context.Dispose() } }
}

function Complete-Entry {
    param([object]$Child,[int]$WaitMilliseconds=120000)
    if($Child.disposed){return $Child.result}
    if(-not $Child.process.WaitForExit($WaitMilliseconds)){$Child.timed_out=$true;throw 'Owned entry host timed out; unknown descendants require retention.'}
    if(-not $Child.stdout_task.Wait(5000) -or -not $Child.stderr_task.Wait(5000)){$Child.timed_out=$true;throw 'Owned entry pipes did not join; preserve scope.'}
    $result=[pscustomobject]@{exit_code=$Child.process.ExitCode;text=$Child.stdout_task.GetAwaiter().GetResult()+[Environment]::NewLine+$Child.stderr_task.GetAwaiter().GetResult()}
    $Child.process.Dispose();$Child.disposed=$true;$Child.result=$result
    $proofs=@($result.text -split '\r?\n'|Where-Object{$_ -match '^\[NATIVEPROOF\] '}|ForEach-Object{$_.Substring(14)|ConvertFrom-Json})
    $audits=@($result.text -split '\r?\n'|Where-Object{$_ -match '^\[ENTRYAUDIT\] '}|ForEach-Object{$_.Substring(13)|ConvertFrom-Json})
    $result|Add-Member -NotePropertyName native_release_proofs -NotePropertyValue $proofs;$result|Add-Member -NotePropertyName audit -NotePropertyValue $audits
    $accountingComplete=$audits.Count -eq 1 -and $audits[0].native_audit_installed -and $audits[0].native_call_count -eq $proofs.Count
    $noNativePreflightProved=$Child.allow_no_native_preflight -and $accountingComplete -and $proofs.Count -eq 0 -and $result.exit_code -eq 1 -and $result.text -match '\[FATAL\].*Preflight failed: File path is 260 UTF-16 units; the supported limit is 259' -and $result.text -match '\[RESULT\] status=FailedBeforeConversion; total=1; processed=0; converted=0; failed=0' -and @(Get-ChildItem -LiteralPath $Child.output_parent -Directory -Force).Count -eq 0
    $Child.descendants_proved=$result.exit_code -in @(0,1,2,130) -and $result.text -match '\[RESULT\] status=' -and $result.text -match '\[ENTRYEND\]' -and $accountingComplete -and (($proofs.Count -gt 0 -and @($proofs|Where-Object{-not $_.process_released}).Count -eq 0) -or $noNativePreflightProved)
    $report.entry_runs += [pscustomobject]@{label=$Child.label;exit_code=$result.exit_code;console=$result.text.Replace($testRoot,'<OWNED_TEMP>');native_release_proofs=@($proofs|ForEach-Object{[pscustomobject]@{process_released=$_.process_released;status=$_.status;exit_code=$_.exit_code;encoder=$_.encoder;output_path=if($_.output_path){$_.output_path.Replace($testRoot,'<OWNED_TEMP>')}else{$null};validation_fault=$_.validation_fault}});audit=$audits;host_wait_exit_and_pipe_join_proved=$true;host_disposed=$true;descendants_proved=$Child.descendants_proved;no_native_preflight_proved=$noNativePreflightProved}
    return $result
}
